function hash(buf) { return crypto.createHash('sha256').update(buf).digest('hex'); }

const linhas = [];
for (const [nome, rel, de, para] of MUTACOES) {
  const arq = path.join(RAIZ, rel);
  const original = fs.readFileSync(arq);
  const texto = original.toString('utf8');
  const n = texto.split(de).length - 1;
  if (n !== 1) { linhas.push(`${nome}: ABORTADA (trecho casou ${n} vezes)`); continue; }
  fs.writeFileSync(arq, texto.replace(de, para));
  const r = spawnSync(process.execPath, ['--test', '--test-force-exit', ...TESTES], { cwd: RAIZ, encoding: 'utf8' });
  fs.writeFileSync(arq, original);
  const restaurado = hash(fs.readFileSync(arq)) === hash(original);
  const falhas = (r.stdout.match(/^ℹ fail (\d+)/m) || [])[1];
  const resultado = r.status === 0 ? 'INERTE (testes passaram)' : `morta (${falhas} teste(s) falharam)`;
  linhas.push(`${nome} [${rel}]: ${resultado}; restaurado byte a byte: ${restaurado ? 'sim' : 'NAO'}`);
}
