phase('Criticar')
const LENTES = [
  'CONCORRÊNCIA E VIRADA: dois aparelhos admitindo o mesmo PR, atribuição perdida, admin caindo no meio de uma atribuição, histerese, relógio, lease, recibo, rodada do dia, dedup por head e postLanes por processo. Monte cenários de linha do tempo concretos.',
  'INVARIANTES E EXCESSO: sincronização desligada sem efeito, um aparelho igual a hoje, versão antiga, invariante 4, work-conserving da justiça de fila, clique manual, e o que é complexidade que não paga (YAGNI). Aponte o que pode ser removido sem perder as decisões do dono.',
]
const criticas = (await parallel(LENTES.map((l, i) => () => agent(
  `${BASE}\n\nMapa (JSON):\n${JSON.stringify(mapa)}\n\nDesenho a derrubar (JSON):\n${JSON.stringify(desenho)}\n\nVocê é um CRÍTICO adversarial com a lente: ${l}\nAbra o código para cada afirmação. Gravidade honesta.`,
  { label: `critica-s3-${i + 1}`, phase: 'Criticar', schema: CRITICA_SCHEMA },
)))).filter(Boolean)

phase('Sintetizar')
const sintese = await agent(
  `${BASE}\n\nSintetize o desenho FINAL da seção 3 a partir do desenho e das críticas. Todo erro fatal ou alto resolvido ou virado risco explícito; aplique os cortes de excesso que não violem decisões do dono; decisoes_para_o_dono só com o que muda o desenho e ainda não foi decidido, cada uma com recomendação.\n\nMapa (JSON):\n${JSON.stringify(mapa)}\n\nDesenho (JSON):\n${JSON.stringify(desenho)}\n\nCríticas (JSON):\n${JSON.stringify(criticas)}`,
  { label: 'sintese-s3', phase: 'Sintetizar', schema: DESENHO_SCHEMA },
)

return { sintese, criticas, desenho, mapa }
