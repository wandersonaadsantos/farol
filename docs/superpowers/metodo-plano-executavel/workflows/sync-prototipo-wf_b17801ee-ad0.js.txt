Seja concreto: arquivo, linha, evidência (saída de comando), correção proposta.`

const FIX_PROMPT = (b, findings) => `
Você corrige achados de revisão do bloco ${b.id} no worktree ${WT} (HEAD destacado). Contrato: ${CONTRATO} (SOMENTE LEITURA). Regras de trabalho idênticas às do implementador:
${BASE}

Achados a corrigir (critical e important; minor só se trivial):
${JSON.stringify(findings, null, 2)}

Para cada achado: corrija com TDD (teste que falha antes, passa depois, quando o achado é de comportamento), rode o gate inteiro verde e faça UM commit por achado: \`git commit -m "fix(sync): <resumo>"\`. Se discordar de um achado, NÃO corrija e explique em deviations com kind "discorda" e a prova. Devolva o schema (tasks = um item por achado corrigido, id = "fix:<task>:<n>").`

const resultados = []
for (const b of BLOCOS) {
  phase('Construir')
  log(`${b.id}: ${b.title}`)
  const impl = await agent(`${BASE}\n\nSEU BLOCO: ${b.id} (${b.title}). Tarefas, nesta ordem:\n${b.tasks}`,
    { label: `impl:${b.id}`, phase: 'Construir', schema: IMPL_SCHEMA, effort: 'high' })
  if (!impl) { resultados.push({ bloco: b.id, erro: 'implementador morreu' }); break }
  if (impl.blocked) { resultados.push({ bloco: b.id, impl, parou: impl.blocked }); break }

  phase('Revisar')
  const rev = await agent(REVIEW_PROMPT(b), { label: `review:${b.id}`, phase: 'Revisar', schema: REVIEW_SCHEMA, effort: 'high' })
  const graves = ((rev && rev.findings) || []).filter(f => f.severity === 'critical' || f.severity === 'important')

  let fix = null
  if (graves.length) {
    phase('Corrigir')
    log(`${b.id}: ${graves.length} achado(s) grave(s), corrigindo`)
    fix = await agent(FIX_PROMPT(b, graves), { label: `fix:${b.id}`, phase: 'Corrigir', schema: IMPL_SCHEMA, effort: 'high' })
  }
  resultados.push({ bloco: b.id, impl, review: rev, fix })
  if (fix && fix.blocked) break
}
return resultados
