phase('Mapear')
const readerRuns = READERS.map(r => () => agent(r.prompt, { label: `map:${r.key}`, phase: 'Mapear', schema: MAP_SCHEMA, effort: 'medium' }).then(v => ({ key: r.key, ...v })))

const auditRuns = AUDITORS.map(a => () => agent(a.prompt, { label: `audit:${a.key}`, phase: 'Mapear', schema: AUDIT_SCHEMA, effort: 'high' }).then(v => ({ key: a.key, ...v })))

const [maps, research, audits] = await Promise.all([
  parallel(readerRuns), parallel(researchRuns), parallel(auditRuns),
])

const findings = audits.filter(Boolean).flatMap(a => (a.findings || []).map(f => ({ ...f, auditor: a.key })))
log(`${findings.length} achados brutos dos auditores; refutando cada um com 3 lentes`)

phase('Refutar')
const VERDICT = { type: 'object', properties: { refuted: { type: 'boolean' }, reason: { type: 'string' }, corrected_fix: { type: 'string' } }, required: ['refuted', 'reason'] }
const LENSES = [
  'lente CÓDIGO: verifique no repositório se o spec de fato deixa isso indefinido ou se o código/CLAUDE.md já responde; refute se já está respondido em algum lugar que o executor lê',
  'lente SPEC: releia a seção citada e as vizinhas; refute se o spec já define isso em outra seção (cite a linha)',
  'lente HAIKU: pergunte se um implementador competente mas sem contexto REALMENTE teria que adivinhar, ou se existe um default óbvio e único; refute se o default é único e seguro',
]
const verified = await pipeline(findings,
  f => parallel(LENSES.map((lens, i) => () => agent(`${COMMON}
Spec: ${SPEC}. Um auditor afirmou a seguinte PONTA SOLTA no spec de sincronização entre dispositivos:
id: ${f.id} | seção: ${f.section} | tipo: ${f.kind}
afirmação: ${f.claim}
evidência: ${f.evidence}
por que o Haiku adivinharia: ${f.why_haiku_would_guess}
correção proposta: ${f.proposed_fix}
Sua tarefa: TENTAR REFUTAR usando a ${lens}. Na dúvida, refuted=true. Se não refutar, opcionalmente melhore a correção em corrected_fix (mais concreta, com nomes/valores).`, { label: `refute:${f.id}:${i}`, phase: 'Refutar', schema: VERDICT, effort: 'medium' }))).then(votes => {
    const v = votes.filter(Boolean)
    const refutedCount = v.filter(x => x.refuted).length
    const survives = v.length > 0 && refutedCount < 2
    const fix = v.filter(x => !x.refuted && x.corrected_fix).map(x => x.corrected_fix)
    return { ...f, survives, votes: v.map(x => ({ refuted: x.refuted, reason: x.reason })), improved_fixes: fix }
  })
)

const confirmed = verified.filter(Boolean).filter(f => f.survives)
const dropped = verified.filter(Boolean).filter(f => !f.survives)
log(`${confirmed.length} pontas soltas confirmadas, ${dropped.length} refutadas`)
return { maps, research, confirmed, dropped: dropped.map(d => ({ id: d.id, claim: d.claim, votes: d.votes })) }
